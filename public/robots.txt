# Allow Google's app-ads.txt crawler explicitly. This is Google's
# recommended defensive setup per
# https://support.google.com/admob/answer/9776740 — even without a
# blocking robots.txt, having an explicit allow removes one variable
# from the AdMob verification troubleshooting flow.
User-agent: Google-adstxt
Disallow:

# Allow Googlebot / general crawling for the rest of the site.
User-agent: *
Allow: /
# Pages are crawlable (their titles, descriptions and share images are built
# on the server); the app's data API is not: rendering a page must not fan out
# into live calls to Overpass / Wikipedia on crawl budgets.
Allow: /api/og/
Disallow: /api/
# Redirects each crawler to wherever its IP geolocates; nothing to index
Disallow: /town/near-me

Sitemap: https://www.go-roam.uk/sitemap.xml
